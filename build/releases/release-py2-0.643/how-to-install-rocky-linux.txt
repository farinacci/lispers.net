
         How to Install a lispers.net Release on Rocky Linux
         ---------------------------------------------------

#
# Get popular use tools.
#
sudo dnf update -y
sudo dnf install -y epel-release
sudo dnf install -y \
    gcc glibc-devel python2 python2-devel python2-pip golang libffi-devel openssl-devel \
    libpcap iptables iproute tcpdump tcsh sudo traceroute iputils net-tools \
    procps-ng emacs vim jq

#
# After python2 install symlink to point 'python' to 'python2'.
#
cd /usr/bin
sudo ln -sf python2 python        

#
# Get lispers.net tarball and put it in its own directory.
#
mkdir lispers.net
cd lispers.net
wget <tarball-url>
tar zxvf lispers.net-release-0.<version>.tgz

#
# Load lispers.net library dependencies.
#
python -m pip install cheroot bottle netifaces pcappy pyOpenSSL \
    pycrypto requests curve25519-donna geopy ecdsa pycryptodome pytun \
    future distro

#
# Special casing tools for pcapy. Need to build libpcap to get pcap.h.
#
set -euo pipefail
sudo dnf install -y dnf-plugins-core
sudo dnf config-manager --set-enabled crb || true
sudo dnf config-manager --set-enabled powertools || true
sudo dnf groupinstall -y "Development Tools"
sudo dnf install -y gcc gcc-c++ make pkgconf-pkg-config \
    python2 python2-devel python2-pip \
    libpcap libpcap-devel
rpm -ql libpcap-devel | egrep '(^/usr/include/pcap.h$|/usr/include/pcap/pcap.h$)' || {
  echo "libpcap headers not found via RPM. Use Option B (source build)."; exit 1; }
python2 -m pip install --upgrade "pip<21" setuptools wheel
CPPFLAGS="-I/usr/include/pcap" python2 -m pip install --no-binary :all: "pcapy==0.11.5"
python2 - <<'PY'
import pcapy; print("pcapy imported OK")
PY
echo "All good."

python -m pip install pcapy
    
#
# Test if install successful.
#
python lispers.net-test-install.pyo

#
# Configure lisp.config and point copy your own custom RL by copying
# RL-template
#
cp RL-template RL
chmod 755 RL
<modify RL file>
<modify lisp.config file>

#
# Start lispers.net LISP subsystem.
#
./RL

#
# Add these to /lispers.net directory.        
#
cat >> ./ver
curl --silent --insecure -u root: https://localhost:8080/lisp/api/data/system | jq .
^D    
cat >> ./eid
curl --silent --insecure -u root: https://localhost:8080/lisp/api/data/database-mapping | jq .
^D
